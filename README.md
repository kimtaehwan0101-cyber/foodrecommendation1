        name = st.text_input("식재료 이름", placeholder="예: 계란, 양파, 김치")
        quantity = st.number_input("수량", min_value=1, value=1, step=1)

    if submitted:

            })

elif menu == "식재료 관리":
    st.header("📦 식재료 관리")
    if not st.session_state.ingredients:
        st.info("등록된 식재료가 없습니다.")
    else:
        for i, item in enumerate(st.session_state.ingredients):
            days = (item["expiry"] - date.today()).days
            a, b, c = st.columns([3, 3, 1])
            a.write(f"**{item['name']}** × {item['quantity']}")
            b.write(f"날짜: {item['expiry']} (D{days:+})")
            if c.button("삭제", key=f"del_{i}"):
                st.session_state.ingredients.pop(i)
                st.rerun()
                elif menu == "음식 추천":
    st.header("🍳 보유 식재료 기반 음식 추천")
    st.caption("등록한 재료와 음식 데이터베이스의 대표 재료를 비교합니다.")
    if not st.session_state.ingredients:
        st.warning("먼저 식재료를 등록해 주세요.")
    else:
        # 이름의 앞뒤 공백을 제거하고 대소문자를 통일해 비교
        owned = {x["name"].strip().casefold() for x in st.session_state.ingredients}
        urgent = {
            x["name"].strip().casefold()
            for x in st.session_state.ingredients
            if 0 <= (x["expiry"] - date.today()).days <= 3
        }

        results = []
        for category, dish, required in RECIPES:
            matched = [food for food in required if food.casefold() in owned]
            if matched:
                coverage = len(matched) / len(required) * 100
                urgent_count = sum(1 for food in matched if food.casefold() in urgent)
                # 유통기한 임박 재료를 많이 활용하는 음식에 우선순위 부여
                priority = coverage + urgent_count * 12
                results.append((priority, category, dish, matched, required, coverage, urgent_count))

        results.sort(key=lambda x: (x[0], len(x[3])), reverse=True)

        only_complete = st.checkbox("필요한 대표 재료를 모두 보유한 음식만 보기", value=False)
        if only_complete:
            results = [r for r in results if len(r[3]) == len(r[4])]

        st.write(f"추천 가능한 음식: **{len(results)}개**")
        if not results:
            st.warning("일치하는 음식이 없습니다. 식재료 이름을 음식 데이터베이스의 재료명과 맞춰 입력해 보세요.")
        else:
            for priority, category, dish, matched, required, coverage, urgent_count in results:
                with st.container(border=True):
                    st.subheader(f"{dish}")
                    st.caption(category)
                    st.write("보유 재료: " + ", ".join(matched))
                    st.write("대표 필요 재료: " + ", ".join(required))
                    st.progress(int(coverage))
                    st.caption(f"대표 재료 보유율: {coverage:.0f}% · 소비기한 임박 재료 활용: {urgent_count}개")
                    if len(matched) == len(required):
                        st.success("데이터베이스에 등록된 대표 재료를 모두 보유하고 있어요.")
                    else:
                        missing = [x for x in required if x not in matched]
                        st.caption("추가로 필요한 대표 재료: " + ", ".join(missing))
